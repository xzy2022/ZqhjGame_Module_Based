

这段不会克隆副本，也不会下载整个 LFS 仓库，只获取指定版本包和离线检查所需资源
```powershell
$Repo = (Resolve-Path '..\ZqhjGame').Path
$Release = 'capture-v31'

Set-Location $Repo

git diff --quiet
if ($LASTEXITCODE -ne 0) {
    throw '仓库存在真实的已跟踪文件修改，请先检查，停止更新。'
}

git diff --cached --quiet
if ($LASTEXITCODE -ne 0) {
    throw '仓库存在暂存修改，请先检查，停止更新。'
}

git fetch origin master
if ($LASTEXITCODE -ne 0) { throw '获取远端更新失败' }

git merge-base --is-ancestor HEAD origin/master
if ($LASTEXITCODE -ne 0) {
    throw '本地与远端不能直接快进，请勿强制覆盖。'
}

git merge --ff-only origin/master
if ($LASTEXITCODE -ne 0) { throw '更新失败' }

$lfsInclude = "artifacts/submission/$Release/**,artifacts/vision/fixtures/**"

git lfs pull --include="$lfsInclude" --exclude=""
if ($LASTEXITCODE -ne 0) { throw '指定版本资源下载失败' }

$packageDir = Join-Path $Repo "artifacts\submission\$Release"
$manifest = Get-Content (Join-Path $packageDir 'manifest.json') -Raw |
    ConvertFrom-Json

$checks = @(
    @{ File = 'agent.py';       Expected = $manifest.agent_sha256 }
    @{ File = 'vision.pt';      Expected = $manifest.vision_sha256 }
    @{ File = 'evaluation.json'; Expected = $manifest.evaluation_sha256 }
)

foreach ($check in $checks) {
    $filePath = Join-Path $packageDir $check.File
    $actual = (Get-FileHash $filePath -Algorithm SHA256).Hash.ToLowerInvariant()
    $expected = ([string]$check.Expected).ToLowerInvariant()

    if ($actual -ne $expected) {
        throw "文件校验失败：$filePath"
    }
}

Write-Host "更新和校验完成：$Release"
git log -1 --oneline
```
